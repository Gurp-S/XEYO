   209→def maybe_advance_aging_boundary(messages: list[dict], working: WorkingSnapshot) -> bool:
   210→	"""非 v61 路径的老化边界推进：把 frozen_until 前移到尾部保护区之前。
   211→
   212→	仅在开启 XEYO_TOOL_AGING 时生效；带滞后门（≥MIN_ADVANCE）与 pair-safe
   213→	切点。**已压缩态（compact_cursor>0）跳过**：避免与 C2 同轮抢推边界、
   214→	制造双 miss。v61 模式不介入（其 C1 由公式决策）。
   215→	"""
   216→	if not aging_enabled() or use_v61():
   217→		return False
   218→	if int(working.compact_cursor or 0) > 0:
   219→		return False
   220→	candidate = pair_safe_cut(messages, keep_tail_cut(messages))
   221→	if candidate - int(working.c1_frozen_until) < _aging_min_advance():
   222→		return False
   223→	note_c1(working, candidate)
   224→	try:
   225→		from audit.log import default_audit_log
   226→
   227→		default_audit_log().record(
   228→			"memory.aging.advance",
   229→			session_id=working.session_id,
   230→			aging_version=AGING_VERSION,
   231→			frozen_until=candidate,
   232→			messages=len(messages),
   233→		)
   234→	except Exception:
   235→		pass
   236→	return True

   239→def idle_seconds(working: WorkingSnapshot) -> float:
   240→	"""根据 last_model_call_at 估算空闲秒数，供 ρ̂ / TTL 先验；无记录则 0"""
   241→	at = working.last_model_call_at
   242→	if at is None:
   243→		return 0.0
   244→	now = datetime.now(at.tzinfo) if at.tzinfo else datetime.now()
   245→	return max(0.0, (now - at).total_seconds())
