    21→# tool_result 内容的最大字符数，超过则截断。从 16k 收紧到 8k（对齐 DSH 的

    29→def _assistant_tool_ids(msg: dict[str, Any]) -> list[str]:
    30→	ids: list[str] = []
    31→	content = msg.get("content")
    32→	if isinstance(content, list):
    33→		for block in content:
    34→			if not isinstance(block, dict) or block.get("type") != "tool_use":
    35→				continue
    36→			uid = str(block.get("id") or "")
    37→			if uid:
    38→				ids.append(uid)
    39→	calls = msg.get("tool_calls")
    40→	if isinstance(calls, list):
    41→		for call in calls:
    42→			if not isinstance(call, dict):
    43→				continue
    44→			uid = str(call.get("id") or "")
    45→			if uid:
    46→				ids.append(uid)
    47→	return ids
    48→

    50→def _tool_result_ids(msg: dict[str, Any]) -> list[str]:
    51→	ids: list[str] = []
    52→	role = msg.get("role")
    53→	if role == "tool":
    54→		tid = str(msg.get("tool_call_id") or "")
    55→		if tid:
    56→			ids.append(tid)
    57→	content = msg.get("content")
    58→	if isinstance(content, list):
    59→		for block in content:
    60→			if not isinstance(block, dict) or block.get("type") != "tool_result":
    61→				continue
    62→			uid = str(block.get("tool_use_id") or "")
    63→			if uid:
    64→				ids.append(uid)
    65→	return ids

    68→def tool_pair_ranges(messages: list[dict[str, Any]]) -> list[tuple[int, int]]:
    69→	"""assistant(tool_calls) → 连续 tool 结果的半开区间列表。"""
    70→	ranges: list[tuple[int, int]] = []
    71→	i = 0
    72→	n = len(messages)
    73→	while i < n:
    74→		ids = _assistant_tool_ids(messages[i])
    75→		if not ids:
    76→			i += 1
    77→			continue
    78→		j = i + 1
    79→		while j < n and _tool_result_ids(messages[j]):
    80→			j += 1
    81→		ranges.append((i, j))
    82→		i = j
    83→	return ranges

    86→def keep_tail_cut(
    87→	messages: list[dict[str, Any]],
    88→	*,
    89→	tool_rounds: int | None = None,
    90→	fallback_messages: int | None = None,
    91→) -> int:
    92→	"""计算尾部保护区起点：优先按最近 ``tool_rounds`` 个成对工具轮。
    93→
    94→	无工具轮时退回 ``len - fallback_messages``（旧 KEEP_TAIL_MESSAGES 行为）。
    95→	返回的下标是「保护区起点」= 可冻结边界候选（不含尾部）。
    96→	"""
    97→	n = len(messages)
    98→	rounds = max(1, int(tool_rounds if tool_rounds is not None else KEEP_TAIL_TOOL_ROUNDS))
    99→	fallback = max(1, int(fallback_messages if fallback_messages is not None else KEEP_TAIL_MESSAGES))
   100→	pairs = tool_pair_ranges(messages)
   101→	if pairs:
   102→		start = pairs[-rounds][0] if len(pairs) >= rounds else pairs[0][0]
   103→		return max(0, min(start, n))
   104→	return max(0, n - fallback)

   106→def split_at_cursor(
   107→	history: list[dict[str, Any]], cursor: int
   108→) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
   109→	"""按 compact_cursor 把历史切成左段/右段；不在本函数里调用 decide"""
   110→	c = max(0, min(int(cursor), len(history)))
   111→	return history[:c], history[c:]
