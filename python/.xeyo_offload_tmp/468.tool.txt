   200→def runtime_mode_snapshot_block(session_id: str) -> str:
   201→	"""审批模式活状态快照（T_now 广播，仅易变维度，turn 首一次）。
   202→
   203→	- #1：**只在 turn 首发一次**（``begin_permission_turn`` 复位后再发），
   204→	  轮内不再注入 —— 避免弱模型在回合中途被新增背景块带偏而重做任务；
   205→	  真开关（门禁）本身已即时生效，不依赖此广播。
   206→	- 头带「（background only）」；措辞含 supersedes，明确最新覆盖此前。
   207→	- 只改投影（copy-on-write），绝不写历史；子代理/side 上下文跳过。
   208→	"""
   209→	if not (session_id or "").strip():
   210→		return ""
   211→	try:
   212→		from permissions.runtime_mode import (
   213→			get_runtime_mode_store,
   214→			runtime_mode_snapshot_text,
   215→		)
   216→
   217→		store = get_runtime_mode_store()
   218→		mode = store.effective(session_id)
   219→		if not mode:
   220→			return ""
   221→		if not store.mark_turn_broadcast(session_id, mode):
   222→			return ""
   223→		return runtime_mode_snapshot_text(mode)
   224→	except Exception:  # noqa: BLE001
