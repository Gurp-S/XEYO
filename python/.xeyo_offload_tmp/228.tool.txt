   165→					turn_id=tid,
   166→					lease_id=lease_id,
   167→					model=model,
   168→					goal_text=(goal_text or "")[:4000],
   169→					user_message_id=user_message_id or "",
   170→					status="running",
   171→				)
   172→				self._turns[session_id] = det
   173→				self._persist(det)
   174→				self._evict_terminal_turns_locked()
   175→			_log.info(
   176→				"turn_start session=%s turn_id=%s reason=submit",
   177→				session_id,
   178→				tid,
   179→			)
   180→			det.task = asyncio.create_task(
   181→				self._run_producer(det, producer),
   182→				name=f"xeyo-turn-{session_id}-{tid}",
   183→			)
   184→			return tid
   185→
   186→	def _evict_terminal_turns_locked(self) -> None:
   187→		"""调用方必须持有 _tlock。终态 turn 只保留最近 N 个会话的，控内存。
   188→
   189→		frames 只用于 reattach 重放；快照已落盘，丢弃旧终态不影响恢复。
   190→		"""
   191→		terminal = [
   192→			(sid, t) for sid, t in self._turns.items() if t.done.is_set()
   193→		]
   194→		if len(terminal) <= _MAX_TERMINAL_TURNS:
   195→			return
   196→		terminal.sort(key=lambda st: st[1].started_at)
   197→		for sid, _t in terminal[: len(terminal) - _MAX_TERMINAL_TURNS]:
   198→			self._turns.pop(sid, None)

   201→		self, det: _DetachedTurn, producer: ProducerFn
   202→	) -> None:
   203→		final_status = "succeeded"
   204→		stop_reason = ""
