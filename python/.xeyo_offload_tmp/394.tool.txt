    21→class PromptAssembler:
    22→    """把 system + 历史拼成模型 API messages。
    23→
    24→
    25→
    26→    会话级 memo：``(cwd, model, tools, custom, append, date, flags, instr_sig)``
    27→
    28→    未变则复用已拼好的左段，避免每轮 submit 重装。
    29→
    30→    """
    31→
    32→    def __init__(self) -> None:
    33→
    34→        self._system_memo: dict[tuple, tuple[str, list[dict]]] = {}
    35→
    36→    def build(self, system: str, history: list[dict]) -> list[dict]:
    37→
    38→        return [{"role": "system", "content": system}, *history]
    39→
    40→    def clear_system_memo(self) -> None:
    41→
    42→        self._system_memo.clear()

    44→    def _memo_key(
    45→
    46→            self,
    47→
    48→            *,
    49→
    50→            cwd: str,
    51→
    52→            model: str,
    53→
    54→            names: Sequence[str],
    55→
    56→            custom_system_prompt: str | None,
    57→
    58→            append_system_prompt: str | None,
    59→
    60→            include_context_blocks: bool,
    61→
    62→            date_iso: str | None,
    63→
    64→    ) -> tuple:
    65→
    66→        from permissions.policy import side_mode as sidemode
    67→
    68→        work = os.path.abspath(os.path.expanduser(cwd or "."))
    69→
    70→        try:
    71→
    72→            from memory.instruction import instruction_cache_signature
    73→
    74→            instr_sig = instruction_cache_signature(work, work)
    75→
    76→        except Exception:

    78→            instr_sig = ()
    79→
    80→        return (
    81→
    82→            work,
    83→
    84→            model,
    85→
    86→            tuple(names),
    87→
    88→            (custom_system_prompt or "").strip(),
    89→
    90→            (append_system_prompt or "").strip(),
    91→
    92→            bool(include_context_blocks),
    93→
    94→            date_iso or "",
    95→
    96→            instr_sig,
    97→
    98→            # side 模式改写左段内容，必须参与 memo 键防串缓存
    99→            sidemode(),
   100→
   101→        )

   103→    async def build_system_parts(
   104→
   105→            self,
   106→
   107→            *,
   108→
   109→            cwd: str,
   110→
   111→            model: str,
   112→
   113→            tools: Any = None,
   114→
   115→            tool_names: Sequence[str] | None = None,
   116→
   117→            custom_system_prompt: str | None = None,
   118→
   119→            append_system_prompt: str | None = None,
   120→
   121→            include_context_blocks: bool = True,
   122→
   123→            date_iso: str | None = None,
   124→
   125→    ) -> tuple[str, list[dict]]:
   126→
   127→        """同 ``build_system`` 但额外返回每段的上下文构成（供用量条分类）。

   132→

   134→
   135→        names = (
   136→
   137→            list(tool_names)
   138→
   139→            if tool_names is not None
   140→
   141→            else tool_names_from_registry(tools)
   142→
   143→        )
   144→
   145→        key = self._memo_key(
   146→
   147→            cwd=cwd,
   148→
   149→            model=model,
   150→
   151→            names=names,
   152→
   153→            custom_system_prompt=custom_system_prompt,
   154→
   155→            append_system_prompt=append_system_prompt,
   156→
   157→            include_context_blocks=include_context_blocks,
   158→
   159→            date_iso=date_iso,
   160→
   161→        )
   162→
   163→        hit = self._system_memo.get(key)
   164→
   165→        if hit is not None:
   166→            return hit
   167→
   168→        parts = await fetch_system_prompt_parts(
   169→
   170→            cwd=cwd,
   171→
   172→            model=model,
   173→
   174→            tool_names=names,
   175→
   176→            custom_system_prompt=custom_system_prompt,
   177→
   178→            date_iso=date_iso,
   179→
   180→        )
   181→
   182→        result = assemble_system_prompt_parts(
   183→
   184→            parts,
   185→
   186→            custom_system_prompt=custom_system_prompt,
   187→
   188→            append_system_prompt=append_system_prompt,
   189→
   190→            include_context_blocks=include_context_blocks,
   191→
   192→        )
   193→
   194→        self._system_memo[key] = result
   195→
   196→        return result
