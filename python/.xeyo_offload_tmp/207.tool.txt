     1→"""会话级工作区上下文（WorkspaceContext）。
     2→
     3→用 contextvars 把 session 与工作目录强绑定，避免多并发会话读取
     4→`session/cwd.py` 的模块级全局变量而串目录。工具/权限应优先读取
     5→`get_workspace_context()` 中的 cwd，而不是模块全局。
     6→"""
     7→
     8→from __future__ import annotations
     9→
    10→import contextvars
    11→from dataclasses import dataclass, field
    12→
    13→
    14→@dataclass
    15→class WorkspaceContext:
    16→	"""一次会话/一件任务的权威工作区上下文。"""
    17→
    18→	session_id: str
    19→	cwd: str
    20→	allowed_paths: list[str] = field(default_factory=list)
    21→	permission_profile: str = "workspace_write"
    22→
    23→
    24→_current: contextvars.ContextVar[WorkspaceContext | None] = contextvars.ContextVar(
    25→	"xeyo_workspace", default=None
    26→)
    27→
    28→
    29→def set_workspace_context(ctx: WorkspaceContext | None) -> None:
    30→	"""在当前 async 上下文设置工作区。None 表示清除。"""
    31→	_current.set(ctx)
    32→
    33→
    34→def get_workspace_context() -> WorkspaceContext | None:
    35→	return _current.get()
    36→
    37→
    38→def get_cwd() -> str:
    39→	"""优先返回当前会话上下文的工作目录；无则回退模块全局。"""
    40→	ctx = _current.get()
    41→	if ctx is not None and ctx.cwd:
    42→		return ctx.cwd
    43→	from session.cwd import get_cwd as _legacy
    44→
    45→	return _legacy(_skip_context=True)
    46→
    47→
