   176→async def _stream_text(client: _StreamClient, prompt: str) -> tuple[str, str | None]:
   177→	"""收集 1-shot 文本；返回 (text, finish_reason)。"""
   178→	from engine.abort import AbortController
   179→

   181→	text = ""
   182→	finish: str | None = None
   183→	async for chunk in client.stream(
   184→		[
   185→			{
   186→				"role": "user",
   187→				"content": prompt,
   188→			}
   189→		],
   190→		[],
   191→		abort,
   192→	):
   193→		kind = getattr(chunk, "kind", "")
   194→		if kind == "text_delta":
   195→			text += str(getattr(chunk, "text", "") or "")

   197→		if fr:
   198→			finish = str(fr)
   199→	return text, finish
