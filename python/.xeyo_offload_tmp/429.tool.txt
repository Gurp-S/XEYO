    36→def ends_with_tool_result(messages: list[dict[str, Any]]) -> bool:
    37→	"""投影末条是否为 tool / 含 tool_result（需 Continue、且勿挂 Memory）。"""
    38→	if not messages:
    39→		return False
    40→	last = messages[-1]
    41→	role = last.get("role")
    42→	if role == "tool":
    43→		return True
    44→	if role != "user":
    45→		return False
    46→	content = last.get("content")
    47→	if not isinstance(content, list):
    48→		return False
    49→	return any(
    50→		isinstance(b, dict) and b.get("type") == "tool_result" for b in content
    51→	)
