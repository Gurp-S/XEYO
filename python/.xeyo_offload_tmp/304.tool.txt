     1→const TOKEN_UNITS = ['', 'k', 'M', 'B', 'T'] as const;
     2→
     3→/** 将 Token 以紧凑且稳定的单位显示，原始值仍由后端保留。 */
     4→export function formatTokenCount(tokens: number): string {
     5→	const value = Math.max(0, Math.round(Number.isFinite(tokens) ? tokens : 0));
     6→	let unitIndex = 0;
     7→	let scaled = value;
     8→	while (scaled >= 1000 && unitIndex < TOKEN_UNITS.length - 1) {
     9→		scaled /= 1000;
    10→		unitIndex += 1;
    11→	}
    12→	if (unitIndex === 0) {
    13→		return `${scaled.toLocaleString('en-US')} tok`;
    14→	}
    15→	const rounded = Number(scaled.toFixed(2));
    16→	if (rounded >= 1000 && unitIndex < TOKEN_UNITS.length - 1) {
    17→		return `${(rounded / 1000).toFixed(2)}${TOKEN_UNITS[unitIndex + 1]} tok`;
    18→	}
    19→	return `${rounded.toFixed(2)}${TOKEN_UNITS[unitIndex]} tok`;
    20→}

    22→export function formatCnyAmount(cny: number | null | undefined): string {
    23→	if (cny == null || !Number.isFinite(cny)) {
    24→		return '费用待确认';
    25→	}
    26→	return `¥${Math.max(0, cny).toFixed(2)}`;
    27→}

    29→/** 顶栏用量芯片：累计消耗 / 模型窗口 · 上下文占用%（窗口未知则省略中段）。 */
    30→export function formatUsageChipPreview(
    31→	consumedTokens: number | null | undefined,
    32→	contextLimit: number | null | undefined,
    33→	contextPercent?: string | null,
    34→): string {
    35→	const left =
    36→		consumedTokens != null && Number.isFinite(consumedTokens)
    37→			? formatTokenCount(consumedTokens)
    38→			: '—';
    39→	const hasWindow =
    40→		contextLimit != null && Number.isFinite(contextLimit) && contextLimit > 0;
    41→	const mid = hasWindow ? formatTokenCount(contextLimit) : null;
    42→	const pct = contextPercent == null ? '占用待确认' : `${contextPercent}%`;
    43→	return mid ? `${left} / ${mid} · ${pct}` : `${left} · ${pct}`;
    44→}

    46→/**
    47→ * 缓存命中率显示（口径对齐 @deepseek-ai/dsh-client-ui-conversation ``cacheHitPercent``，
    48→ * 显示精度收敛为一位小数）：「计费输入 = 命中 + 未命中(含写档)」。
    49→ *
    50→ * - 常规：一位小数（例 85.2）
    51→ * - 一位小数已四舍五入到 100（但并非全命中）→ 不谎报 100，按真实分位给高精度 99.9x
    52→ * - 全命中 → 100；无计费输入 → null（显示"暂无数据"）
    53→ *
    54→ * - hit：会话累计命中（cacheHitTokens，DeepSeek = prompt_cache_hit_tokens）
    55→ * - miss：会话累计未命中（cacheMissTokens；DeepSeek 无独立 write 档，
    56→ *   未命中已含写入部分 → 与 DSH 的 uncached+write 口径一致）
    57→ */
    58→export function formatCacheHitPercent(
    59→	hit: number,
    60→	miss: number,
    61→): string | null {
    62→	const read = Math.max(0, Number.isFinite(hit) ? hit : 0);
    63→	const writeAndUncached = Math.max(0, Number.isFinite(miss) ? miss : 0);
    64→	const denominator = read + writeAndUncached;
    65→	if (denominator === 0) return null;
    66→	if (writeAndUncached === 0) return '100';
    67→	const oneDecimal = (read / denominator * 100).toFixed(1);
    68→	if (oneDecimal !== '100.0') return oneDecimal;
    69→	// 一位小数已撞 100% 但并非全命中：取 DSH 的高精度分位（99.9x），不谎报 100。
    70→	return nearHundredPrecision(writeAndUncached, denominator);
    71→}
    72→
    73→/** DSH 同款：99.9x 高精度分位（复制自 dsh-client-ui-conversation cacheHitPercent 尾部）。 */
    74→function nearHundredPrecision(missedTokens: number, denominator: number): string {
    75→	let decimalPlaces = 1;
    76→	let scaledDoubleGap = missedTokens * 200;
    77→	const denominatorTens = Math.floor(denominator / 10);
    78→	while (scaledDoubleGap <= denominatorTens) {
    79→		scaledDoubleGap *= 10;
    80→		decimalPlaces += 1;
    81→	}
    82→	const denominatorOnes = denominator % 10;
    83→	let roundedLoss = 5;
    84→	for (let loss = 1; loss < 5; loss += 1) {
    85→		const factor = loss * 2 + 1;
    86→		const threshold = factor * denominatorTens + Math.floor((factor * denominatorOnes) / 10);
    87→		if (scaledDoubleGap <= threshold) {
    88→			roundedLoss = loss;
    89→			break;
    90→		}
    91→	}
    92→	return `99.${'9'.repeat(decimalPlaces - 1)}${10 - roundedLoss}`;
    93→}
