   115→def project(
   116→    history: list[dict[str, Any]], *, frozen_until: int = 0
   117→) -> list[dict[str, Any]]:
   118→    """投影 history：C0 截断 + C1 占位（frozen_until 之前的 tool_result 一律占位）。
   119→
   120→    不修改入参。未改动的消息与入参共享引用（copy-on-write）；仅 C0/C1
   121→    实际改写的消息分配新对象。frozen_until 由 runtime 按公式决策推进，
   122→    本函数只做 Apply，因此两次 C1/C2 之间旧消息的投影字节保持不变。
   123→    """
   124→    id_to_name = build_tool_use_names(history)
   125→    boundary = max(0, min(int(frozen_until), len(history)))
   126→    aging = aging_enabled()
   127→    return [
   128→        _process_message(msg, idx, boundary, id_to_name, aging=aging)
   129→        for idx, msg in enumerate(history)
   130→    ]
   131→
   132→
   133→def build_tool_use_names(history: list[dict[str, Any]]) -> dict[str, str]:
   134→    """从历史记录中提取所有 assistant 消息中的 tool_use 块，建立 tool_use_id -> 工具名映射。"""
   135→    return _collect_tool_use_names(history, {})
   136→
   137→
   138→def _collect_tool_use_names(
   139→    history: list[dict[str, Any]], into: dict[str, str]
   140→) -> dict[str, str]:
   141→    """把消息里的 tool_use id → name 增量并入映射（供增量投影复用前缀映射）。"""
   142→    for msg in history:
   143→        if msg.get("role") != "assistant":
   144→            continue
   145→        content = msg.get("content")
   146→        if not isinstance(content, list):
   147→            continue
   148→        for block in content:
   149→            if not isinstance(block, dict) or block.get("type") != "tool_use":
   150→                continue
   151→            uid = block.get("id")
   152→            if uid:
   153→                into[str(uid)] = str(block.get("name") or "tool")
   154→    return into

   157→def _project_tool_result_content(
   158→    raw: str,
   159→    *,
   160→    uid: str,
   161→    name: str,
   162→    frozen: bool,
   163→    frozen_until: int,
   164→    idx: int,
   165→    aging: bool,
   166→    is_error: bool,
   167→) -> tuple[str, bool]:
   168→    """对单个 tool_result 内容做 C0/C1，返回 (投影文本, 是否写入老化存根)。"""
   169→    n_lines = raw.count("\n") + (1 if raw else 0)
   170→    from prompt.fence import (
   171→        truncate_tool_content_preserving_fence,
   172→        unwrap_tool_output,
   173→    )
   174→
   175→    inner, _fname = unwrap_tool_output(raw)
   176→    if len(inner) > MAX_TOOL_RESULT_CHARS:
   177→        raw = truncate_tool_content_preserving_fence(
   178→            raw,
   179→            max_chars=MAX_TOOL_RESULT_CHARS,
   180→            marker="\n…[truncated]…\n",
   181→            fallback_suffix=TRUNCATE_SUFFIX,
   182→        )
   183→
   184→    aged = False
   185→    if frozen:
   186→        if aging:
   187→            if not should_exempt(is_error, name):
   188→                before = len(raw)
   189→                raw = build_stub(
   190→                    name,
   191→                    uid,
   192→                    raw.split("\n", 1)[0],
   193→                    folded=(frozen_until - idx) > FOLD_AFTER,
   194→                )
   195→                record_stub(before, len(raw))
   196→                aged = True
   197→        else:
   198→            raw = (
   199→                f"[compacted] {name}: prior result ({n_lines} lines) "
   200→                "archived; answer from remaining context"
   201→            )
   202→    return raw, aged
