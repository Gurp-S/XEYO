   773→def run_pre_llm_inject(

   783→	# 预算（F1）。新增块必须声明 KLASS_*，遗漏按 INVENTORY 处理（最保守）。

   802→		tagged.append((KLASS_DIRECTIVE, WRAP_UP_INSTRUCTIONS))

   805→			(KLASS_DIRECTIVE, f"# Runtime budget notice\n{ctx.runtime_notice}")
   806→		)
   807→	# 批次1：仅工具续写轮注入——fresh-user 轮里它是上一个任务的推理残留，

   811→			(
   812→				KLASS_DIRECTIVE,
   813→				"# 上一轮思考回顾（截选）\n"
   814→				"你在上一轮模型调用中已经推理过，结尾如下。这是延续，不是新任务；"

   822→
   823→			tagged.append((KLASS_DIRECTIVE, MULTI_AGENT_HINT.strip()))
   824→		except Exception:
   825→			_log.debug("MULTI_AGENT_HINT load failed", exc_info=True)
   826→
   827→	# T6：重复调用递进提醒（background only）——query_loop 轮内状态，

   831→		from engine.repeat_guard import current_advice
   832→
   833→		rep = current_advice()
   834→		if rep and not ctx.subagent:
   835→			tagged.append(
   836→				(KLASS_DIRECTIVE, f"# Repeat guard（background only）\n{rep}")
   837→			)
   838→	except Exception:
   839→		_log.debug("repeat advice inject failed", exc_info=True)
   840→

   844→		ctx.instructions_enabled() and ctx.working is not None and (ctx.cwd or "").strip()

   846→	if allow_instr and ctx.working is not None:
   847→		cwd = ctx.cwd.strip()
   848→		new_paths = _discover_nested_from_projection(out, ctx.working, cwd)

   854→		else:

   857→				out, tool_names=frozenset({"Write", "Edit", "NotebookEdit"})

   859→			nested_block = _format_nested_block(
   860→				_nested_paths_for_touched(
   861→					list(ctx.working.loaded_nested_instruction_paths or []),
   862→					recent_touched,
   863→				)
   864→			)
   865→		if nested_block:
   866→			tagged.append((KLASS_INVENTORY, nested_block))

   871→
   872→			change = nested_change_notice(ctx.working, commit=False)
   873→			if change:
   874→				tagged.append((KLASS_INVENTORY, change))
   875→				pending_nested_change = True
   876→		except Exception:
   877→			_log.debug("nested change notice failed", exc_info=True)

   881→				from memory.instruction_maintain import stale_instruction_notice

   883→				# commit=False：确认进最终 blocks 后再刷 stamp
   884→				stale = stale_instruction_notice(cwd, commit=False)
   885→				if stale:
   886→					tagged.append((KLASS_DIRECTIVE, stale))
   887→					pending_stale_commit_cwd = cwd
   888→			except Exception:
   889→				_log.debug("stale inject failed", exc_info=True)
   890→		# 批次1：Proposals digest 不再推送 T_now——模型对候选晋升无可执行
   891→		# 动作（NightShift / 人工确认），纯 ambient 噪音。拉取通道：
   892→		# /proposals slash 命令；Memory(action=search) 结果附带候选计数行

   905→		tagged.append((KLASS_DIRECTIVE, compact_block))

   914→
   915→			mcp_warn = mcp_required_warning_block(ctx.cwd.strip())

   917→				tagged.append((KLASS_EVENT, mcp_warn))
   918→		except Exception:
   919→			_log.debug("mcp required warning inject failed", exc_info=True)

   921→	# F2.5：reconcile push/pull 活页块（# 工具面变更 / # 技能目录变更）。

   924→		try:
   925→			from extension.reconcile import consume_reconcile_blocks
   926→
   927→			for blk in consume_reconcile_blocks():
   928→				if blk and str(blk).strip():

   954→				ctx.cwd.strip(), ctx.session_id.strip(), touched

  1002→	if not after_tools and _is_vague_referent_turn(out):
  1003→		tagged = [(k, t) for k, t in tagged if k != KLASS_INVENTORY]

  1006→	kept = _trim_tagged_blocks(tagged)
