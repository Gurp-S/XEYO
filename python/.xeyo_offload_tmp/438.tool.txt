    34→def default_sessions_dir() -> Path:
    35→    """默认目录: ~/.xeyo/sessions/（可用 XEYO_SESSIONS_DIR 覆盖）。"""
    36→    override = os.environ.get("XEYO_SESSIONS_DIR", "").strip()
    37→    if override:
    38→        return Path(override).expanduser()
    39→    return Path.home() / ".xeyo" / "sessions"

    42→def safe_session_filename(session_id: str) -> str:
    43→    """会话键 → 稳定文件名：``:`` → ``__``，其余非法字符 → ``_``。"""
    44→    raw = session_id or ""
    45→    parts: list[str] = []
    46→    for ch in raw:
    47→        if ch == ":":
    48→            parts.append("__")
    49→        elif _SAFE_CHAR.match(ch):
    50→            parts.append(ch)
    51→        else:
    52→            parts.append("_")
    53→    name = "".join(parts).strip("._") or "session"
    54→    if len(name) > _MAX_FILENAME:
    55→        digest = hashlib.sha256(raw.encode("utf-8")).hexdigest()[:16]
    56→        name = name[:160].rstrip("._-") + "_" + digest
    57→    return name
