     1→/**
     2→ * usageAccumulator.ts — per-stream usage accounting, extracted verbatim from
     3→ * streamSendSlice.createStreamSendSlice() flushUsage/scheduleUsage/pendingUsage.
     4→ * Behavior unchanged; the onUsage stream handler delegates to push().
     5→ */
     6→import type {StoreApi} from 'zustand';
     7→import {
     8→	type CompressionStreamEvent,
     9→	type UsageStreamEvent,
    10→} from '@/lib/api';
    11→import {
    12→	saveSession,
    13→} from '@/lib/db';
    14→import {
    15→	sessionStreamActive,
    16→} from '@/lib/sessionStreams';
    17→import {
    18→	emitPastureEvent,
    19→} from '@/pasture/events/PastureEvents';
    20→import {
    21→	type ChatState,
    22→} from './preStoreHelpers';
    23→
    24→type SetState = StoreApi<ChatState>['setState'];
    25→type GetState = StoreApi<ChatState>['getState'];
    26→
    27→export type UsageAccumulator = {
    28→	flush: () => void;
    29→	schedule: () => void;
    30→	/** append a usage event and schedule an rAF flush (guard handled by caller). */
    31→	push: (ev: UsageStreamEvent) => void;
    32→	/** full onUsage dispatch: guard + pasture context snapshot + flush schedule. */
    33→	onUsage: (ev: UsageStreamEvent) => void;
    34→	/** full onCompression dispatch: pasture compression start/complete snapshot. */
    35→	onCompression: (ev: Omit<CompressionStreamEvent, 'kind'>) => void;
    36→};
    37→
    38→export function createUsageAccumulator(
    39→	get: GetState,
    40→	set: SetState,
    41→	sessionId: string,
    42→): UsageAccumulator {
    43→	let usageRaf = 0;
    44→	let pendingUsage: UsageStreamEvent[] = [];
    45→
    46→	const flush = () => {
    47→		if (usageRaf) {
    48→			cancelAnimationFrame(usageRaf);
    49→			usageRaf = 0;
    50→		}
    51→		if (pendingUsage.length === 0 || !sessionStreamActive(get(), sessionId)) {
    52→			pendingUsage = [];
    53→			return;
    54→		}
    55→		const events = pendingUsage;
    56→		pendingUsage = [];
    57→		const current = get();
    58→		const previous = current.sessionUsageById[sessionId];
    59→		let nextUsage = previous;
    60→		for (const ev of events) {
    61→			const hasContextTokens = typeof ev.contextTokens === 'number';
    62→			const hasContextLimit = typeof ev.contextLimit === 'number' && ev.contextLimit > 0;
    63→			const estimated = (
    64→				nextUsage?.costSource === 'unknown' ||
    65→				ev.costSource === 'unknown'
    66→			);
    67→			nextUsage = {
    68→				promptTokens: (nextUsage?.promptTokens ?? 0) + ev.promptTokens,
    69→				completionTokens: (nextUsage?.completionTokens ?? 0) + ev.completionTokens,
    70→				cacheHitTokens: (nextUsage?.cacheHitTokens ?? 0) + ev.cacheHitTokens,
    71→				cacheMissTokens: (nextUsage?.cacheMissTokens ?? 0) + ev.cacheMissTokens,
    72→				tokens: (nextUsage?.tokens ?? 0) + ev.tokens,
    73→				cny: (nextUsage?.cny ?? 0) + ev.cny,
    74→				requests: (nextUsage?.requests ?? 0) + 1,
    75→				costSource:
    76→					estimated
    77→						? 'unknown'
    78→						: nextUsage?.costSource === 'api' && ev.costSource === 'api'
    79→						? 'api'
    80→						: 'estimate',
    81→				usdLimit: ev.usdLimit,
    82→				contextTokens: hasContextTokens ? ev.contextTokens : nextUsage?.contextTokens,
    83→				// smoke-test #2：事件未带窗口时清空而非沿用旧值 —— 换模型后
    84→				// 旧窗口不会残留（显示"窗口未知"比显示错误窗口更诚实）。
    85→				contextLimit: hasContextLimit ? ev.contextLimit : undefined,
    86→				contextPercent:
    87→					hasContextTokens && hasContextLimit
    88→						? Math.min(100, Math.max(0, (ev.contextTokens! / ev.contextLimit!) * 100))
    89→						: nextUsage?.contextPercent,
    90→				contextSource: hasContextTokens ? 'stream' : nextUsage?.contextSource,
    91→				dataQuality: hasContextTokens && hasContextLimit ? 'measured' : nextUsage?.dataQuality,
    92→				lastContextAt: hasContextTokens ? Date.now() : nextUsage?.lastContextAt,
    93→				contextBreakdown: ev.contextBreakdown?.length
    94→					? ev.contextBreakdown
    95→					: nextUsage?.contextBreakdown,
    96→				// 单轮（非累计）拆分：与 contextTokens 同轮，供「上下文构成」回退条
    97→				// 使用；累计值会超过窗口，不能拿来画本轮构成。
    98→				lastCacheHitTokens: ev.cacheHitTokens,
    99→				lastCacheMissTokens: ev.cacheMissTokens,
   100→				lastCompletionTokens: ev.completionTokens,
   101→				compactCursor:
   102→					typeof ev.compactCursor === 'number'
   103→						? ev.compactCursor
   104→						: nextUsage?.compactCursor,
   105→				lastAction: ev.lastAction || nextUsage?.lastAction,
   106→				c2SummaryChars:
   107→					typeof ev.c2SummaryChars === 'number'
   108→						? ev.c2SummaryChars
   109→						: nextUsage?.c2SummaryChars,
   110→			};
   111→		}
   112→		if (!nextUsage) {
   113→			return;
   114→		}
   115→		const nextSession = current.sessions.find(s => s.id === sessionId);
   116→		if (nextSession) {
   117→			const persisted = {...nextSession, usage: nextUsage, updatedAt: Date.now()};
   118→			void saveSession(persisted);
   119→			set(state => ({
   120→				sessions: state.sessions.map(s => s.id === sessionId ? persisted : s),
   121→				sessionUsageById: {...state.sessionUsageById, [sessionId]: nextUsage},
   122→			}));
   123→		} else {
   124→			set(state => ({
   125→				sessionUsageById: {...state.sessionUsageById, [sessionId]: nextUsage},
   126→			}));
   127→		}
   128→	};
   129→
   130→	const schedule = () => {
   131→		if (!usageRaf) {
   132→			usageRaf = requestAnimationFrame(flush);
   133→		}
   134→	};
   135→
   136→	const push = (ev: UsageStreamEvent) => {
   137→		pendingUsage.push(ev);
   138→		schedule();
   139→	};
   140→
   141→	const onUsage = (ev: UsageStreamEvent) => {
   142→		if (!sessionStreamActive(get(), sessionId)) {
   143→			return;
   144→		}
   145→		if (typeof ev.contextTokens === 'number' || typeof ev.contextLimit === 'number') {
   146→			emitPastureEvent({
   147→				type: 'CONTEXT_UPDATE',
   148→				sessionId,
   149→				snapshot: {
   150→					currentTokens: ev.contextTokens,
   151→					contextLimit: ev.contextLimit,
   152→					usagePercent:
   153→						typeof ev.contextTokens === 'number' &&
   154→						typeof ev.contextLimit === 'number' && ev.contextLimit > 0
   155→							? (ev.contextTokens / ev.contextLimit) * 100
   156→							: undefined,
   157→					contextSource: 'stream',
   158→					dataQuality: 'measured',
   159→					lastUpdatedAt: Date.now(),
   160→					isStale: false,
   161→				},
   162→			});
   163→		}
   164→		push(ev);
   165→	};
   166→
   167→	const onCompression = (ev: Omit<CompressionStreamEvent, 'kind'>) => {
   168→		if (!sessionStreamActive(get(), sessionId)) return;
   169→		const snapshot = {
   170→			currentTokens: ev.contextTokens,
   171→			contextLimit: ev.contextLimit,
   172→			usagePercent:
   173→				typeof ev.contextTokens === 'number' &&
   174→				typeof ev.contextLimit === 'number' && ev.contextLimit > 0
   175→					? (ev.contextTokens / ev.contextLimit) * 100
   176→					: undefined,
   177→			contextSource: 'stream' as const,
   178→			dataQuality: 'measured' as const,
   179→			lastUpdatedAt: Date.now(),
   180→			isStale: false,
   181→		};
   182→		if (ev.phase === 'start') {
   183→			emitPastureEvent({type: 'CONTEXT_COMPRESSION_START', sessionId, source: ev.source});
   184→		} else {
   185→			emitPastureEvent({
   186→				type: 'CONTEXT_COMPRESSION_COMPLETE',
   187→				sessionId,
   188→				source: ev.source,
   189→				snapshot,
   190→			});
   191→		}
   192→	};
   193→
   194→	return {flush, schedule, push, onUsage, onCompression};
   195→}
