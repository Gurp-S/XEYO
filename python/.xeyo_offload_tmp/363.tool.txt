    92→	if not raw:
    93→		return DEFAULT_RHO_AGE_TABLE
    94→	out: list[tuple[float, float]] = []
    95→	for row in raw:
    96→		age, s = row[0], row[1]
    97→		age_f = float("inf") if age in ("inf", None) else float(age)
    98→		out.append((age_f, float(s)))
    99→	out.sort(key=lambda x: x[0])
   100→	return tuple(out)
