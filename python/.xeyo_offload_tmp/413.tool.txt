  1576→def _c2_pressure_ratio(working: WorkingSnapshot | None, params=None, window_override: int | None = None) -> float:

  1581→	  窗口大时与硬顶严重错位）。

  1592→	if not _c2_formula_enabled("XEYO_C2_PRESSURE_FORMULA"):

  1594→	# 显式 env 覆盖（A/B 定参脚本用）：优先级最高，绕过窗口校验（脚本可无窗口直接注入比值）。
  1595→	raw = os.environ.get("XEYO_C2_PRESSURE_RATIO", "").strip()

  1597→		try:
  1598→			v = float(raw)
  1599→			if 0.0 <= v <= 1.0:
  1600→				return v
  1601→		except ValueError:
  1602→			pass
  1603→	try:
  1604→		from memory.simulator.params import load_params as _lp

  1608→
  1609→		# 真实模型窗口：**必须**由上层传入（用户添加模型时必填的上下文窗口）。
  1610→		# 不再回退 params.window_tokens（恒 128k）——那会让 C2 误以为窗口只有 128k。
  1611→		# 无真实窗口 → 返回 None（未知），调用方按「窗口未知 → 不触发压力 C2」处理（安全）。
  1612→		win = int(window_override) if window_override and int(window_override) > 0 else None
  1613→		if win is None:

  1616→		rounds = _c2_retain_rounds()
